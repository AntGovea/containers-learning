# Create a docker registry in GitHub Container Registry

## create variable with classic access tocken

´´´

  $env:CR_PAT="my_classsic_token"

´´´

## login docker with github using my username 

´´´

  echo $env:CR_PAT | docker login ghcr.io -u AntGovea --password-stdin

´´´
## create container registry in github packages from local image
´´´

  docker push ghcr.io/antgovea/teslo-shop:1.2.0

´´´
## deploy of image with multi-platform on github packages using dockerfile with new tag
´´´

  docker buildx build  --platform linux/amd64,linux/arm64  -t ghcr.io/antgovea/teslo-shop:1.3.0 --push .

´´´
